/ Lazy loading
IntersectionObserver dispara la carga de cada escena 3D solo cuando está a 200px del viewport. Las 7 pizzas nunca cargan en simultáneo.
/ LOD adaptativo
Modelos en 3 niveles: high (desktop), mid (tablet), low (mobile). Swap basado en devicePixelRatio y memoria disponible.
/ Instanced rendering
Toppings repetidos (pepperoni, aceitunas) usan InstancedMesh. Una sola draw call para 50+ instancias.
/ Pause when offscreen
renderer.setAnimationLoop se pausa al cambiar de sección. requestVideoFrameCallback para sync con scroll.
/ Preload + prefetch
Hero preload completo. Siguiente pizza prefetch. CDN edge con HTTP/3 + Brotli.

// AURUM — PizzaScene component
// Cada pizza hereda este wrapper. El comportamiento se customiza vía `config`.

import { useRef, useEffect, useMemo } from 'react'
import * as THREE from 'three'
import { gsap } from 'gsap'
import { ScrollTrigger } from 'gsap/ScrollTrigger'
import { useGLTF } from '@react-three/drei'
import { CheeseStretchMaterial } from '@/shaders/CheeseStretch'
import { SteamVolumetric } from '@/shaders/SteamVolumetric'

gsap.registerPlugin(ScrollTrigger)

export function PizzaScene({ config, isActive }) {
  const groupRef = useRef()
  const meshRef = useRef()
  const knifeRef = useRef()
  const tlRef = useRef()

  // Carga lazy del modelo según pizza
  const { nodes } = useGLTF(`/models/${config.slug}.glb`)

  // Timeline de animación — disparada por scroll
  useEffect(() => {
    if (!groupRef.current || !meshRef.current) return

    const tl = gsap.timeline({
      scrollTrigger: {
        trigger: groupRef.current.parentElement,
        start: 'top 60%',
        end:   'bottom 20%',
        scrub: 1.2,
      }
    })

    // Fase 1 — pizza aparece con stagger de toppings
    tl.from(groupRef.current.scale, {
      x: 0, y: 0, z: 0,
      duration: 0.6,
      ease: 'expo.out',
    })

    // Fase 2 — cuchillo entra
    tl.from(knifeRef.current.position, {
      y: 2,
      duration: 0.4,
      ease: 'power3.in',
    }, '+=0.1')

    // Fase 3 — corte (config.behavior define la animación específica)
    config.behavior({ mesh: meshRef.current, knife: knifeRef.current, tl })

    // Fase 4 — slice se separa
    tl.to('.slice-group', {
      y: 0.3,
      rotationZ: 0.15,
      duration: 0.3,
      ease: 'power2.out',
    })

    tlRef.current = tl
    return () => tl.kill()
  }, [config])

  // Hover — microinteracción de rotación
  const handlePointerMove = (e) => {
    const { offsetX, offsetY } = e.nativeEvent
    const rotY = (offsetX / window.innerWidth - 0.5) * 0.6
    const rotX = (offsetY / window.innerHeight - 0.5) * 0.3
    gsap.to(groupRef.current.rotation, {
      y: rotY, x: rotX,
      duration: 0.8,
      ease: 'power2.out',
    })
  }

  return (
    <group ref={groupRef} onPointerMove={handlePointerMove}>
      <mesh ref={meshRef} geometry={doughGeometry} receiveShadow>
        <meshStandardMaterial color="#e8c89a" roughness={0.7} />
      </mesh>

      <mesh position={[0, 0.01, 0]}>
        <circleGeometry args={[1.15, 64]} />
        <cheeseStretchMaterial uStretch={0.0} uTime={0} />
      </mesh>

      {config.toppings.map((t, i) => (
        <InstancedTopping key={i} {...t} count={t.count} />
      ))}

      <SteamVolumetric position={[0, 0.3, 0]} active={isActive} />

      <group ref={knifeRef} position={[0, 0, 0]}>
        <mesh position={[0, 1.5, 0]}>
          <boxGeometry args={[0.04, 0.8, 0.18]} />
          <meshStandardMaterial color="#d4d4d0" metalness={0.9} roughness={0.2} />
        </mesh>
      </group>

      <spotLight position={[3, 5, 2]} angle={0.4} penumbra={0.6} intensity={2} color="#ffffff" />
      <spotLight position={[-3, 3, -1]} angle={0.5} penumbra={0.8} intensity={0.8} color="#56dddb" />
      <ambientLight intensity={0.4} />
    </group>
  )
}

// Behavior específico de cada pizza — inyecto en config
export const pizzaBehaviors = {
  pepperoni: ({ mesh, knife, tl }) => {
    tl.to(knife.position, { y: 0, duration: 0.3 }, 'cut')
      .call(() => spawnPepperoniPhysics(mesh), [], 'cut')
  },
  bbq: ({ tl }) => {
    tl.to('.cheese-material', { uStretch: 1.0, duration: 0.4 })
  },
  // ... napolitana, caprese, works, cordillerana, vegetariana
}
